Explanation
This Solidity program is a Student Data Management smart contract that stores and retrieves student information on the blockchain. It uses a struct called Student to store the roll number, name, age, and course of each student, and a dynamic array students to store multiple student records. The addStudent() function adds a new student to the array, while getStudent() retrieves the details of a student using their index. getStudentCount() returns the total number of students. The contract also demonstrates Solidity's receive and fallback functions for handling Ether sent to the contract. The receive() function is called when Ether is sent without any data, while the fallback() function is called when unknown function data is sent. The counters receiveCallCount and fallbackCallCount record how many times these functions are triggered, and totalEtherReceived records the total Ether received in Wei.
Important Functions and Concepts
- struct Student – Defines the data fields for each student.
- students[] – Dynamic array that stores all student records.
- addStudent() – Adds a new student to the blockchain.
- getStudent(index) – Retrieves student details using an index.
- getStudentCount() – Returns the number of stored students.
- require() – Checks that the requested student index is valid.
- receive() – Automatically executes when Ether is sent without transaction data.
- fallback() – Executes when unknown data is sent to the contract.
- msg.value – Contains the amount of Ether received.
- view – Indicates that a function only reads data and does not modify the blockchain.
- recordEther() – Adds received Ether to totalEtherReceived.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract StudentData {

    // Structure to store student details
    struct Student {
        uint rollNo;
        string name;
        uint age;
        string course;
    }

    // Dynamic array to store student records
    Student[] public students;

    uint public fallbackCallCount;
    uint public receiveCallCount;
    uint public totalEtherReceived;

    // Function to add student details
    function addStudent(
        uint _rollNo,
        string memory _name,
        uint _age,
        string memory _course
    ) public {
        students.push(Student(_rollNo, _name, _age, _course));
    }

    // Function to get student details
    function getStudent(uint index)
        public
        view
        returns (uint, string memory, uint, string memory)
    {
        require(index < students.length, "Invalid student index");

        Student memory s = students[index];

        return (s.rollNo, s.name, s.age, s.course);
    }

    // Function to get total number of students
    function getStudentCount() public view returns (uint) {
        return students.length;
    }

    // Internal function to record received Ether
    function recordEther() internal {
        totalEtherReceived += msg.value;
    }

    // Receive function: called when Ether is sent without data
    receive() external payable {
        receiveCallCount++;
        recordEther();
    }

    // Fallback function: called when unknown data is sent
    fallback() external payable {
        fallbackCallCount++;
        recordEther();
    }
}
